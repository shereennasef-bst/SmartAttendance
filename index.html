    <script>

        // إعدادات Firebase

        // ملاحظة: القيم يتم تمريرها تلقائياً من بيئة التشغيل

        const firebaseConfig = JSON.parse(__firebase_config);

        firebase.initializeApp(firebaseConfig);

        const db = firebase.firestore();

        const auth = firebase.auth();

        const appId = typeof __app_id !== 'undefined' ? __app_id : 'omama-attendance-system';

 

        const PERIOD_NAMES = { "1":"الأولى", "2":"الثانية", "3":"الثالثة", "4":"الرابعة", "5":"الخامسة", "6":"السادسة", "7":"السابعة" };

        const GRADE_CONFIG = {

            "10": { name: "الصف العاشر", classes: 6 },

            "11-sci": { name: "11 علمي", classes: 4 },

            "11-lit": { name: "11 أدبي", classes: 2 },

            "12-sci": { name: "12 علمي", classes: 4 },

            "12-lit": { name: "12 أدبي", classes: 2 }

        };

 

        let currentData = { subject: '', teacher: '', names: Array(15).fill('') };

        let allCloudData = [];

        let saveTimer;

 

        // عند تحميل الصفحة

        window.onload = function() {

            const today = new Date();

            const year = today.getFullYear();

            const month = String(today.getMonth() + 1).padStart(2, '0');

            const day = String(today.getDate()).padStart(2, '0');

            document.getElementById('dateInput').value = `${year}-${month}-${day}`;

            initAuth();

        };

 

        async function initAuth() {

            try {

                if (typeof __initial_auth_token !== 'undefined' && __initial_auth_token) {

                    await auth.signInWithCustomToken(__initial_auth_token);

                } else {

                    await auth.signInAnonymously();

                }

                document.getElementById('sync-indicator').innerHTML = `<div class="w-2 h-2 rounded-full bg-green-500"></div> متصل بالسحابة ✅`;

                listenToCloud();

            } catch (e) {

                document.getElementById('sync-indicator').innerHTML = `⚠️ عطل في المزامنة`;

            }

        }

 

        function listenToCloud() {

            db.collection('artifacts').doc(appId).collection('public').doc('data').collection('attendance')

                .onSnapshot(snap => {

                    allCloudData = snap.docs.map(doc => doc.data());

                    if (!document.getElementById('mainForm').classList.contains('hidden')) {

                        loadAttendance();

                    }

                }, () => {

                    document.getElementById('sync-indicator').innerHTML = `⚠️ خطأ اتصال`;

                });

        }

 

        function updateClassOptions() {

            const grade = document.getElementById('gradeSelect').value;

            const container = document.getElementById('classContainer');

            const select = document.getElementById('classSelect');

           

            if (!grade) {

                container.classList.add('hidden');

                document.getElementById('mainForm').classList.add('hidden');

                document.getElementById('placeholder').classList.remove('hidden');

                return;

            }

 

            container.classList.remove('hidden');

            const config = GRADE_CONFIG[grade];

            let html = `<option value="">اختر الفصل...</option>`;

            for(let i=1; i<=config.classes; i++) {

                const label = grade.startsWith('10') ? `10-${i}` : (grade.includes('sci') ? `${grade.split('-')[0]}ع${i}` : `${grade.split('-')[0]}أ${i}`);

                html += `<option value="${i}">${label}</option>`;

            }

            select.innerHTML = html;

        }

 

        function loadAttendance() {

            const date = document.getElementById('dateInput').value;

            const grade = document.getElementById('gradeSelect').value;

            const classNum = document.getElementById('classSelect').value;

            const period = document.getElementById('periodSelect').value;

 

            if(!date || !grade || !classNum) return;

 

            document.getElementById('placeholder').classList.add('hidden');

            document.getElementById('mainForm').classList.remove('hidden');

 

            const id = `${date}_G${grade}_P${period}_C${classNum}`;

            const existing = allCloudData.find(d => d.id === id);

 

            if (existing) {

                currentData = {

                    subject: existing.subject || '',

                    teacher: existing.teacher || '',

                    names: [...(existing.names || Array(15).fill(''))]

                };

            } else {

                currentData = { subject: '', teacher: '', names: Array(15).fill('') };

            }

 

            document.getElementById('subjectInput').value = currentData.subject;

            document.getElementById('teacherInput').value = currentData.teacher;

            renderNames();

        }

 

        function renderNames() {

            const grid = document.getElementById('namesGrid');

            grid.innerHTML = '';

            for(let i=0; i<15; i++) {

                grid.innerHTML += `

                    <div class="relative">

                        <span class="absolute right-3 top-1/2 -translate-y-1/2 text-[10px] font-black text-slate-300">${i+1}</span>

                        <input type="text" value="${currentData.names[i] || ''}"

                            oninput="updateName(${i}, this.value)"

                            placeholder="اسم الطالبة..."

                            class="w-full pr-10 pl-4 py-3 bg-slate-50 rounded-xl border-2 border-transparent focus:border-blue-500 outline-none font-bold text-sm">

                    </div>

                `;

            }

            updateTotal();

        }

 

        function updateName(index, val) {

            currentData.names[index] = val;

            updateTotal();

            autoSave();

        }

 

        function updateTotal() {

            const count = currentData.names.filter(n => n && n.trim() !== "").length;

            document.getElementById('totalBadge').innerText = count;

        }

 

        function autoSave() {

            clearTimeout(saveTimer);

            document.getElementById('sync-indicator').innerHTML = `<div class="w-2 h-2 rounded-full bg-orange-500 animate-pulse"></div> جاري الحفظ...`;

           

            saveTimer = setTimeout(async () => {

                const date = document.getElementById('dateInput').value;

                const grade = document.getElementById('gradeSelect').value;

                const classNum = document.getElementById('classSelect').value;

                const period = document.getElementById('periodSelect').value;

               

                currentData.subject = document.getElementById('subjectInput').value;

                currentData.teacher = document.getElementById('teacherInput').value;

 

                const id = `${date}_G${grade}_P${period}_C${classNum}`;

               

                try {

                    await db.collection('artifacts').doc(appId).collection('public').doc('data').collection('attendance').doc(id).set({

                        ...currentData,

                        id, date, grade, classNum, period

                    });

                    document.getElementById('sync-indicator').innerHTML = `<div class="w-2 h-2 rounded-full bg-green-500"></div> تم الحفظ سحابياً ✅`;

                } catch(e) {

                    document.getElementById('sync-indicator').innerHTML = `⚠️ فشل الحفظ`;

                }

            }, 800);

        }

        function exportFullReport() {

            const date = document.getElementById('dateInput').value;

            const wb = XLSX.utils.book_new();

 

            Object.keys(GRADE_CONFIG).forEach(gradeKey => {

                const config = GRADE_CONFIG[gradeKey];

                const gradeData = allCloudData.filter(d => d.date === date && d.grade === gradeKey);

                if(gradeData.length === 0) return;

 

                const rows = [];

                rows.push([`سجل حصر الغياب اليومي ثانوية أمامة بنت بشر`, "", "", "", "", "", "التاريخ:", date]);

                rows.push([config.name]);

                rows.push([]);

 

                let totalGradeAbsence = 0;

 

                for(let p=1; p<=7; p++) {

                    const periodData = gradeData.filter(d => d.period == p);

                    if(periodData.length === 0) continue;

 

                    rows.push([`الحصة ${PERIOD_NAMES[p]}`]);

                    const classRow = ["الفصل"];

                    const subRow = ["المادة"];

                    const teachRow = ["المعلمة"];

                   

                    for(let c=1; c<=config.classes; c++) {

                        const label = gradeKey.startsWith('10') ? `10-${c}` : (gradeKey.includes('sci') ? `${gradeKey.split('-')[0]}ع${c}` : `${gradeKey.split('-')[0]}أ${c}`);

                        classRow.push(label);

                        const d = periodData.find(x => x.classNum == c) || {subject:'', teacher:''};

                        subRow.push(d.subject);

                        teachRow.push(d.teacher);

                    }

                    rows.push(classRow, subRow, teachRow);

 

                    for(let i=0; i<15; i++) {

                        const nameRow = [i+1];

                        for(let c=1; c<=config.classes; c++) {

                            const d = periodData.find(x => x.classNum == c);

                            nameRow.push(d && d.names[i] ? d.names[i] : "");

                        }

                        rows.push(nameRow);

                    }

                   

                    const totalPeriodRow = ["إجمالي الحصة"];

                    let periodTotal = 0;

                    for(let c=1; c<=config.classes; c++) {

                        const d = periodData.find(x => x.classNum == c);

                        const count = d ? d.names.filter(n => n && n.trim() !== "").length : 0;

                        totalPeriodRow.push(count);

                        periodTotal += count;

                    }

                    totalPeriodRow.push("المجموع للحصة:", periodTotal);

                    rows.push(totalPeriodRow, []);

                    totalGradeAbsence += periodTotal;

                }

 

                rows.push([]);

                rows.push(["الإجمالي العام لغياب المرحلة (كل الحصص):", totalGradeAbsence]);

                rows.push([""]);

                rows.push(["إعداد المعلمة: شيرين ناصف"]);

                rows.push(["مع تحيات قسم الحاسوب"]);

                rows.push(["مديرة المدرسة: أ. عائشة الكندري"]);

 

                const ws = XLSX.utils.aoa_to_sheet(rows);

                XLSX.utils.book_append_sheet(wb, ws, config.name);

            });

 

            XLSX.writeFile(wb, `غياب_أمامة_بنت_بشر_${date}.xlsx`);

        }
